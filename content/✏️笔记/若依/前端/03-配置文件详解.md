## 三、vite.config.js 配置文件

### 3.1 文件位置
`d:\WorkSpace\cloned\java\RuoYi-Vue3\vite.config.js`

### 3.2 完整配置解析

```javascript
import { defineConfig, loadEnv } from 'vite'
import path from 'path'
import createVitePlugins from './vite/plugins'

// 后端接口地址
const baseUrl = 'http://localhost:8080'

export default defineConfig(({ mode, command }) => {
  const env = loadEnv(mode, process.cwd())
  const { VITE_APP_ENV } = env
  
  return {
    // ==================== 基础配置 ====================
    // 部署路径，默认根路径
    base: VITE_APP_ENV === 'production' ? '/' : '/',
    
    // 注册插件
    plugins: createVitePlugins(env, command === 'build'),
    
    // ==================== 路径解析 ====================
    resolve: {
      // 路径别名
      alias: {
        '~': path.resolve(__dirname, './'),    // 项目根目录
        '@': path.resolve(__dirname, './src')  // src 目录
      },
      // 文件扩展名
      extensions: ['.mjs', '.js', '.ts', '.jsx', '.tsx', '.json', '.vue']
    },
    
    // ==================== 构建配置 ====================
    build: {
      sourcemap: command === 'build' ? false : 'inline',  // Source Map
      outDir: 'dist',              // 输出目录
      assetsDir: 'assets',         // 静态资源目录
      chunkSizeWarningLimit: 2000, // 代码块大小警告阈值
      rollupOptions: {
        output: {
          // 代码块文件名格式
          chunkFileNames: 'static/js/[name]-[hash].js',
          // 入口文件名格式
          entryFileNames: 'static/js/[name]-[hash].js',
          // 静态资源文件名格式
          assetFileNames: 'static/[ext]/[name]-[hash].[ext]'
        }
      }
    },
    
    // ==================== 开发服务器 ====================
    server: {
      port: 80,        // 端口号
      host: true,      // 监听所有地址
      open: true,      // 自动打开浏览器
      
      // 代理配置（解决跨域）
      proxy: {
        // 主接口代理
        '/dev-api': {
          target: baseUrl,           // 后端地址
          changeOrigin: true,        // 改变请求源
          rewrite: (p) => p.replace(/^\/dev-api/, '')  // 重写路径
        },
        // Swagger 文档代理
        '^/v3/api-docs/(.*)': {
          target: baseUrl,
          changeOrigin: true
        }
      }
    },
    
    // ==================== CSS 配置 ====================
    css: {
      postcss: {
        plugins: [
          {
            // 移除 charset 声明（避免编码问题）
            postcssPlugin: 'internal:charset-removal',
            AtRule: {
              charset: (atRule) => {
                if (atRule.name === 'charset') {
                  atRule.remove()
                }
              }
            }
          }
        ]
      }
    }
  }
})
```

### 3.3 关键配置项说明

| 配置项 | 说明 | 默认值 |
|--------|------|--------|
| `base` | 部署路径 | `/` |
| `resolve.alias` | 路径别名 | `@` → `./src` |
| `server.port` | 开发服务器端口 | `80` |
| `server.proxy` | API 代理配置 | 代理到 `localhost:8080` |
| `build.outDir` | 打包输出目录 | `dist` |

## 五、Vite 插件配置

### 5.1 插件入口文件

**文件位置**：`vite/plugins/index.js`

```javascript
import vue from '@vitejs/plugin-vue'
import createAutoImport from './auto-import'
import createSvgIcon from './svg-icon'
import createCompression from './compression'
import createSetupExtend from './setup-extend'

export default function createVitePlugins(viteEnv, isBuild = false) {
  const vitePlugins = [vue()]  // Vue 插件必须
  
  vitePlugins.push(createAutoImport())     // 自动导入
  vitePlugins.push(createSetupExtend())    // setup 扩展
  vitePlugins.push(createSvgIcon(isBuild)) // SVG 图标
  
  // 仅在构建时启用压缩
  isBuild && vitePlugins.push(...createCompression(viteEnv))
  
  return vitePlugins
}
```

### 5.2 自动导入配置

**文件位置**：`vite/plugins/auto-import.js`

```javascript
import autoImport from 'unplugin-auto-import/vite'

export default function createAutoImport() {
  return autoImport({
    imports: [
      'vue',           // 自动导入 Vue API
      'vue-router',    // 自动导入 Vue Router API
      'pinia'          // 自动导入 Pinia API
    ],
    dts: false  // 不生成类型声明文件
  })
}
```

**效果**：无需手动导入，可直接使用：
```javascript
// 无需 import { ref, computed } from 'vue'
const count = ref(0)        // 自动识别
const double = computed(() => count.value * 2)  // 自动识别
```

### 5.3 SVG 图标配置

**文件位置**：`vite/plugins/svg-icon.js`

```javascript
import { createSvgIconsPlugin } from 'vite-plugin-svg-icons'
import path from 'path'

export default function createSvgIcon(isBuild) {
  return createSvgIconsPlugin({
    // 指定需要缓存的图标文件夹
    iconDirs: [path.resolve(process.cwd(), 'src/assets/icons/svg')],
    // 指定 symbolId 格式
    symbolId: 'icon-[dir]-[name]',
    // 是否注入到 HTML
    inject: 'body-last',
    // 自定义插入位置
    customDomId: '__svg__icons__dom__'
  })
}
```

### 5.4 构建压缩配置

**文件位置**：`vite/plugins/compression.js`

```javascript
import compression from 'vite-plugin-compression'

export default function createCompression(env) {
  const plugin = []
  // Gzip 压缩
  if (env.VITE_BUILD_COMPRESS?.includes('gzip')) {
    plugin.push(
      compression({
        ext: '.gz',
        algorithm: 'gzip',
        threshold: 10240  // 大于 10KB 才压缩
      })
    )
  }
  // Brotli 压缩
  if (env.VITE_BUILD_COMPRESS?.includes('brotli')) {
    plugin.push(
      compression({
        ext: '.br',
        algorithm: 'brotliCompress',
        threshold: 10240
      })
    )
  }
  return plugin
}
```

## 八、配置修改指南

### 8.1 修改后端接口地址

修改 `vite.config.js` 中的 `baseUrl`：

```javascript
const baseUrl = 'http://your-backend-server:port'
```

### 8.2 修改开发服务器端口

修改 `vite.config.js` 中的 `server.port`：

```javascript
server: {
  port: 3000,  // 改为你需要的端口
}
```

### 8.3 修改部署路径

如果部署到子路径（如 `https://example.com/admin/`）：

```javascript
// vite.config.js
base: VITE_APP_ENV === 'production' ? '/admin/' : '/'
```

### 8.4 添加新的环境变量

1. 在 `.env.development` 或 `.env.production` 中添加：
```bash
VITE_APP_CUSTOM_VAR = 'custom-value'
```

2. 在代码中使用：
```javascript
const customVar = import.meta.env.VITE_APP_CUSTOM_VAR
```

**注意**：环境变量必须以 `VITE_` 开头才能被 Vite 识别！
