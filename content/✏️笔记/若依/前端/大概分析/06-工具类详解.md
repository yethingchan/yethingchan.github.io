### 2.2 parseTime - 日期格式化

```javascript
/**
 * 日期格式化
 * @param {Date|String|Number} time - 时间
 * @param {String} pattern - 格式模板，默认 '{y}-{m}-{d} {h}:{i}:{s}'
 * @returns {String} 格式化后的日期字符串
 */
export function parseTime(time, pattern) {
  if (!time) return null
  
  const format = pattern || '{y}-{m}-{d} {h}:{i}:{s}'
  let date
  
  if (typeof time === 'object') {
    date = time
  } else {
    // 字符串数字转整数
    if ((typeof time === 'string') && (/^[0-9]+$/.test(time))) {
      time = parseInt(time)
    }
    // 10位时间戳转13位
    if ((typeof time === 'number') && (time.toString().length === 10)) {
      time = time * 1000
    }
    date = new Date(time)
  }
  
  const formatObj = {
    y: date.getFullYear(),
    m: date.getMonth() + 1,
    d: date.getDate(),
    h: date.getHours(),
    i: date.getMinutes(),
    s: date.getSeconds(),
    a: date.getDay()
  }
  
  const time_str = format.replace(/{(y|m|d|h|i|s|a)+}/g, (result, key) => {
    let value = formatObj[key]
    if (key === 'a') { 
      return ['日', '一', '二', '三', '四', '五', '六'][value] 
    }
    if (result.length > 0 && value < 10) {
      value = '0' + value
    }
    return value || 0
  })
  
  return time_str
}
```

**使用示例**：
```javascript
// 默认格式
parseTime(new Date())  // '2024-01-15 10:30:45'

// 自定义格式
parseTime(new Date(), '{y}年{m}月{d}日')  // '2024年01月15日'

// 时间戳
parseTime(1705299445000)  // '2024-01-15 10:30:45'

// 只获取日期
parseTime(new Date(), '{y}-{m}-{d}')  // '2024-01-15'
```

### 2.4 addDateRange - 添加日期范围

```javascript
/**
 * 添加日期范围参数
 * @param {Object} params - 查询参数对象
 * @param {Array} dateRange - 日期范围 [开始日期, 结束日期]
 * @param {String} propName - 属性名前缀
 * @returns {Object} 处理后的参数对象
 */
export function addDateRange(params, dateRange, propName) {
  let search = params
  search.params = typeof search.params === 'object' ? search.params : {}
  dateRange = Array.isArray(dateRange) ? dateRange : []
  
  if (typeof propName === 'undefined') {
    search.params['beginTime'] = dateRange[0]
    search.params['endTime'] = dateRange[1]
  } else {
    search.params['begin' + propName] = dateRange[0]
    search.params['end' + propName] = dateRange[1]
  }
  
  return search
}
```

**使用示例**：
```javascript
const queryParams = {
  pageNum: 1,
  pageSize: 10
}
const dateRange = ['2024-01-01', '2024-01-31']

// 添加日期范围
addDateRange(queryParams, dateRange)
// 结果: { pageNum: 1, pageSize: 10, params: { beginTime: '2024-01-01', endTime: '2024-01-31' } }

// 自定义属性名
addDateRange(queryParams, dateRange, 'CreateTime')
// 结果: { pageNum: 1, pageSize: 10, params: { beginCreateTime: '2024-01-01', endCreateTime: '2024-01-31' } }
```

### 2.5 selectDictLabel - 回显字典标签

```javascript
/**
 * 回显数据字典
 * @param {Array} datas - 字典数据数组
 * @param {String|Number} value - 字典值
 * @returns {String} 字典标签
 */
export function selectDictLabel(datas, value) {
  if (value === undefined) return ""
  
  const actions = []
  Object.keys(datas).some((key) => {
    if (datas[key].value == ('' + value)) {
      actions.push(datas[key].label)
      return true
    }
  })
  
  if (actions.length === 0) {
    actions.push(value)
  }
  
  return actions.join('')
}
```

**使用示例**：
```javascript
const sexDict = [
  { label: '男', value: '0' },
  { label: '女', value: '1' }
]

selectDictLabel(sexDict, '0')  // '男'
selectDictLabel(sexDict, '1')  // '女'
selectDictLabel(sexDict, '2')  // '2' (未找到返回原值)
```

### 2.6 handleTree - 构造树形结构

```javascript
/**
 * 构造树型结构数据
 * @param {Array} data - 数据源
 * @param {String} id - id字段，默认 'id'
 * @param {String} parentId - 父节点字段，默认 'parentId'
 * @param {String} children - 孩子节点字段，默认 'children'
 * @returns {Array} 树形数据
 */
export function handleTree(data, id, parentId, children) {
  const config = {
    id: id || 'id',
    parentId: parentId || 'parentId',
    childrenList: children || 'children'
  }
  
  const childrenListMap = {}
  const tree = []
  
  // 初始化
  for (const d of data) {
    const id = d[config.id]
    childrenListMap[id] = d
    if (!d[config.childrenList]) {
      d[config.childrenList] = []
    }
  }
  
  // 构建树
  for (const d of data) {
    const parentId = d[config.parentId]
    const parentObj = childrenListMap[parentId]
    
    if (!parentObj) {
      tree.push(d)  // 根节点
    } else {
      parentObj[config.childrenList].push(d)  // 子节点
    }
  }
  
  return tree
}
```

**使用示例**：
```javascript
const data = [
  { id: 1, parentId: 0, name: '一级部门' },
  { id: 2, parentId: 1, name: '二级部门A' },
  { id: 3, parentId: 1, name: '二级部门B' },
  { id: 4, parentId: 2, name: '三级部门' }
]

const tree = handleTree(data)
// 结果:
// [
//   {
//     id: 1, parentId: 0, name: '一级部门',
//     children: [
//       { id: 2, parentId: 1, name: '二级部门A', children: [{ id: 4, parentId: 2, name: '三级部门', children: [] }] },
//       { id: 3, parentId: 1, name: '二级部门B', children: [] }
//     ]
//   }
// ]
```

### 3.2 debounce - 防抖函数

```javascript
/**
 * 防抖函数
 * @param {Function} func - 要防抖的函数
 * @param {Number} wait - 等待时间(ms)
 * @param {Boolean} immediate - 是否立即执行
 * @returns {Function} 防抖后的函数
 */
export function debounce(func, wait, immediate) {
  let timeout, args, context, timestamp, result
  
  const later = function() {
    const last = +new Date() - timestamp
    if (last < wait && last > 0) {
      timeout = setTimeout(later, wait - last)
    } else {
      timeout = null
      if (!immediate) {
        result = func.apply(context, args)
        if (!timeout) context = args = null
      }
    }
  }
  
  return function(...args) {
    context = this
    timestamp = +new Date()
    const callNow = immediate && !timeout
    if (!timeout) timeout = setTimeout(later, wait)
    if (callNow) {
      result = func.apply(context, args)
      context = args = null
    }
    return result
  }
}
```

**使用示例**：
```javascript
// 搜索输入防抖
const handleSearch = debounce((keyword) => {
  // 执行搜索
  getList(keyword)
}, 300)

// 窗口大小变化防抖
const handleResize = debounce(() => {
  // 重新计算布局
}, 200)
```

### 3.3 deepClone - 深拷贝

```javascript
/**
 * 深拷贝
 * @param {Object|Array} source - 源对象
 * @returns {Object|Array} 拷贝后的对象
 */
export function deepClone(source) {
  if (!source && typeof source !== 'object') {
    throw new Error('error arguments', 'deepClone')
  }
  
  const targetObj = source.constructor === Array ? [] : {}
  
  Object.keys(source).forEach(keys => {
    if (source[keys] && typeof source[keys] === 'object') {
      targetObj[keys] = deepClone(source[keys])
    } else {
      targetObj[keys] = source[keys]
    }
  })
  
  return targetObj
}
```

**使用示例**：
```javascript
const original = { a: 1, b: { c: 2 } }
const cloned = deepClone(original)

cloned.b.c = 999
console.log(original.b.c)  // 2 (原对象不受影响)
```

## 五、字典工具 (dict.js)

**文件位置**：`src/utils/dict.js`

```javascript
import useDictStore from '@/store/modules/dict'
import { getDicts } from '@/api/system/dict/data'

/**
 * 获取字典数据
 * @param {...String} args - 字典类型
 * @returns {Object} 响应式字典数据
 */
export function useDict(...args) {
  const res = ref({})
  
  return (() => {
    args.forEach((dictType, index) => {
      res.value[dictType] = []
      
      // 先从缓存获取
      const dicts = useDictStore().getDict(dictType)
      if (dicts) {
        res.value[dictType] = dicts
      } else {
        // 从后端获取
        getDicts(dictType).then(resp => {
          res.value[dictType] = resp.data.map(p => ({
            label: p.dictLabel,
            value: p.dictValue,
            elTagType: p.listClass,
            elTagClass: p.cssClass
          }))
          // 缓存到 store
          useDictStore().setDict(dictType, res.value[dictType])
        })
      }
    })
    return toRefs(res.value)
  })()
}
```

```vue

      v-for="dict in sys_normal_disable"
```
