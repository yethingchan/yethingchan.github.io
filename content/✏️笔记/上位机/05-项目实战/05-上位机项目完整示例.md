# 05 - 上位机项目完整示例

> 本文通过一个完整的汇川PLC数据监控与控制系统，展示如何将前面学到的所有知识整合到一个实际项目中。

## 一、项目需求描述

### 1.1 项目背景

某工厂有一台汇川PLC控制温度、压力和流量三个关键参数。需要开发一个上位机监控系统，实现以下功能：

### 1.2 功能需求

| 模块 | 功能描述 |
|------|----------|
| **通信管理** | 通过Modbus TCP连接汇川PLC，支持自动重连 |
| **数据采集** | 定时读取PLC数据（温度D100、压力D102、流量D104） |
| **数据显示** | 实时显示当前值，绘制趋势曲线图 |
| **数据存储** | 所有采集数据保存到SQLite数据库 |
| **报警管理** | 数据超限时触发报警，支持报警确认和历史查询 |
| **参数设置** | 可配置PLC连接参数、采集周期、报警阈值 |
| **数据查询** | 查询历史采集数据，支持按时间范围筛选和导出 |
| **数据备份** | 自动备份SQLite数据库，支持手动备份和恢复 |

### 1.3 技术选型

| 技术项 | 选择 | 说明 |
|--------|------|------|
| 开发框架 | .NET Framework 4.8 + WinForms | 工控电脑兼容性好 |
| 通信协议 | Modbus TCP | 通过NModbus库实现 |
| 数据库 | SQLite | 轻量级嵌入式数据库 |
| 日志框架 | Serilog | 结构化日志 |
| 配置管理 | JSON配置文件 | Newtonsoft.Json |
| ORM | Dapper | 轻量级，性能好 |

### 3.2 通信模块实现

```csharp
// ============================================================
// 文件：ModbusTcpCommunication.cs
// 说明：Modbus TCP通信实现
// ============================================================
using HwPlcMonitor.Core.Interfaces;
using HwPlcMonitor.Core.Models;
using HwPlcMonitor.Core.Logging;
using NModbus;
using System.Net.Sockets;

namespace HwPlcMonitor.Core.Communication
{
    /// <summary>
    /// Modbus TCP通信实现
    /// 实现ICommunicationService接口
    /// 支持连接、断开、读写和自动重连
    /// </summary>
    public class ModbusTcpCommunication : ICommunicationService
    {
        private readonly ILogger _log = LogCategories.Communication;
        private TcpClient _tcpClient;
        private IModbusMaster _master;
        private readonly DeviceConfig _config;

        // 重连相关
        private System.Timers.Timer _reconnectTimer;
        private bool _disposed = false;

        public bool IsConnected { get; private set; }

        public event EventHandler<bool> ConnectionStateChanged;
        public event EventHandler<string> CommunicationError;

        public ModbusTcpCommunication(DeviceConfig config)
        {
            _config = config;

            // 初始化重连定时器
            _reconnectTimer = new System.Timers.Timer(_config.ReconnectIntervalS * 1000);
            _reconnectTimer.Elapsed += async (s, e) => await TryReconnectAsync();
        }

        /// <summary>连接到PLC</summary>
        public async Task<bool> ConnectAsync()
        {
            try
            {
                _log.Information("正在连接PLC {Ip}:{Port}, 站号: {SlaveId}",
                    _config.IpAddress, _config.Port, _config.SlaveId);

                _tcpClient = new TcpClient();
                // 设置连接超时
                var connectTask = _tcpClient.ConnectAsync(_config.IpAddress, _config.Port);
                if (await Task.WhenAny(connectTask, Task.Delay(_config.TimeoutMs)) != connectTask)
                {
                    throw new TimeoutException("连接超时");
                }

                // 创建Modbus主站
                var factory = new ModbusFactory();
                _master = factory.CreateMaster(_tcpClient);
                _master.Transport.ReadTimeout = _config.TimeoutMs;
                _master.Transport.WriteTimeout = _config.TimeoutMs;
                _master.Transport.Retries = 3;

                IsConnected = true;
                ConnectionStateChanged?.Invoke(this, true);
                _reconnectTimer.Stop(); // 连接成功停止重连
                _log.Information("PLC连接成功 {Ip}:{Port}", _config.IpAddress, _config.Port);

                return true;
            }
            catch (Exception ex)
            {
                IsConnected = false;
                CommunicationError?.Invoke(this, $"连接失败: {ex.Message}");
                _log.Error(ex, "PLC连接失败");
                return false;
            }
        }

        /// <summary>断开连接</summary>
        public async Task DisconnectAsync()
        {
            try
            {
                IsConnected = false;
                _reconnectTimer.Stop();
                _tcpClient?.Close();
                _master = null;
                ConnectionStateChanged?.Invoke(this, false);
                _log.Information("PLC已断开连接");
            }
            catch (Exception ex)
            {
                _log.Error(ex, "断开连接时发生错误");
            }
        }

        /// <summary>
        /// 读取单个值（从指定地址读取一个寄存器）
        /// 汇川PLC通过Modbus映射寄存器：
        ///   D区 -> 功能码03（读保持寄存器），起始地址 = D编号 + 偏移
        /// </summary>
        public bool ReadValue(string address, out object value)
        {
            value = null;
            if (!IsConnected || _master == null) return false;

            try
            {
                // 解析地址（如 "D100" -> 地址100）
                var (startAddress, functionCode) = ParseAddress(address);

                // 读取1个寄存器（16位）
                var data = _master.ReadHoldingRegisters(
                    (byte)_config.SlaveId, startAddress, 1);

                // 假设数据为16位无符号整数，根据实际PLC配置可能需要缩放
                value = data[0];
                return true;
            }
            catch (Exception ex)
            {
                CommunicationError?.Invoke(this, $"读取失败 {address}: {ex.Message}");
                _log.Warning(ex, "读取寄存器失败 {Address}", address);
                return false;
            }
        }

        /// <summary>
        /// 批量读取多个地址的值（性能优化：一次请求读取多个寄存器）
        /// 前提条件：地址必须是连续的，否则需要分多次读取
        /// </summary>
        public Dictionary<string, object> ReadMultipleValues(List<string> addresses)
        {
            var result = new Dictionary<string, object>();
            if (!IsConnected || _master == null || addresses.Count == 0)
                return result;

            try
            {
                // 解析所有地址，找到起始地址和结束地址
                var parsedAddresses = addresses
                    .Select(a => (Name: a, Address: ParseAddress(a).StartAddress))
                    .OrderBy(a => a.Address)
                    .ToList();

                var startAddr = parsedAddresses.First().Address;
                var endAddr = parsedAddresses.Last().Address;
                var count = endAddr - startAddr + 1;

                _log.Debug("批量读取: 起始地址={Start}, 数量={Count}", startAddr, count);

                // 一次读取所有寄存器
                var data = _master.ReadHoldingRegisters(
                    (byte)_config.SlaveId, startAddr, count);

                // 将读取的数据映射到各地址
                foreach (var pa in parsedAddresses)
                {
                    var index = pa.Address - startAddr;
                    if (index < data.Length)
                        result[pa.Name] = data[index];
                }

                return result;
            }
            catch (Exception ex)
            {
                CommunicationError?.Invoke(this, $"批量读取失败: {ex.Message}");
                _log.Warning(ex, "批量读取失败");
                return result; // 返回空结果
            }
        }

        /// <summary>
        /// 向指定地址写入值
        /// </summary>
        public bool WriteValue(string address, object value)
        {
            if (!IsConnected || _master == null) return false;

            try
            {
                var (startAddress, _) = ParseAddress(address);
                var writeValue = Convert.ToUInt16(value);

                _master.WriteSingleRegister(
                    (byte)_config.SlaveId, startAddress, writeValue);

                _log.Information("写入寄存器 {Address} = {Value}", address, writeValue);
                return true;
            }
            catch (Exception ex)
            {
                CommunicationError?.Invoke(this, $"写入失败 {address}: {ex.Message}");
                _log.Error(ex, "写入寄存器失败 {Address} = {Value}", address, value);
                return false;
            }
        }

        /// <summary>
        /// 解析地址字符串
        /// 汇川PLC Modbus地址映射规则：
        ///   D0 -> 地址0, D100 -> 地址100
        /// </summary>
        private (int StartAddress, byte FunctionCode) ParseAddress(string address)
        {
            address = address.ToUpper().Trim();

            if (address.StartsWith("D"))
            {
                var numStr = address.Substring(1);
                return (int.Parse(numStr), 0x03); // 功能码03：读保持寄存器
            }

            throw new ArgumentException($"不支持的地址格式: {address}");
        }

        /// <summary>尝试自动重连</summary>
        private async Task TryReconnectAsync()
        {
            if (IsConnected || _disposed) return;
            _log.Information("尝试自动重连...");
            await ConnectAsync();
        }

        public void Dispose()
        {
            _disposed = true;
            _reconnectTimer?.Stop();
            _reconnectTimer?.Dispose();
            _tcpClient?.Close();
        }
    }
}
```

```csharp
using System.Data.SQLite;

namespace HwPlcMonitor.Data
{
    /// <summary>
    /// 数据库帮助类
    /// 负责数据库文件的创建、表结构的初始化
    /// </summary>
    public static class DatabaseHelper
    {
        private static string _connectionString;

        /// <summary>获取数据库连接字符串</summary>
        public static string ConnectionString => _connectionString;

        /// <summary>
        /// 初始化数据库
        /// 如果数据库文件不存在则创建，并初始化表结构
        /// </summary>
        public static void Initialize(string dbFilePath)
        {
            // 确保数据库目录存在
            var dbDir = Path.GetDirectoryName(dbFilePath);
            if (!string.IsNullOrEmpty(dbDir) && !Directory.Exists(dbDir))
                Directory.CreateDirectory(dbDir);

            _connectionString = $"Data Source={dbFilePath};Version=3;Pooling=True;Max Pool Size=10;";

            // 判断是否需要创建表
            var isNewDb = !File.Exists(dbFilePath);

            using (var conn = new SQLiteConnection(_connectionString))
            {
                conn.Open();
                if (isNewDb)
                    CreateTables(conn);
            }
        }

        /// <summary>创建数据库表</summary>
        private static void CreateTables(SQLiteConnection conn)
        {
            using (var cmd = conn.CreateCommand())
            {
                // 采集数据表
                cmd.CommandText = @"
                    CREATE TABLE IF NOT EXISTS plc_data (
                        id            INTEGER PRIMARY KEY AUTOINCREMENT,
                        tag_name      TEXT    NOT NULL,
                        address       TEXT    NOT NULL,
                        value         REAL    NOT NULL,
                        unit          TEXT,
                        quality       INTEGER DEFAULT 0,
                        collect_time  TEXT    NOT NULL
                    );
                    CREATE INDEX IF NOT EXISTS idx_plc_data_time ON plc_data(collect_time);
                    CREATE INDEX IF NOT EXISTS idx_plc_data_tag  ON plc_data(tag_name);
                ";
                cmd.ExecuteNonQuery();

                // 报警记录表
                cmd.CommandText = @"
                    CREATE TABLE IF NOT EXISTS alarm_record (
                        id                INTEGER PRIMARY KEY AUTOINCREMENT,
                        tag_name          TEXT    NOT NULL,
                        alarm_description TEXT,
                        trigger_value     REAL,
                        threshold         REAL,
                        alarm_time        TEXT    NOT NULL,
                        recover_time      TEXT,
                        is_acknowledged   INTEGER DEFAULT 0,
                        ack_time          TEXT,
                        alarm_level       TEXT
                    );
                    CREATE INDEX IF NOT EXISTS idx_alarm_time ON alarm_record(alarm_time);
                ";
                cmd.ExecuteNonQuery();
            }
        }

        /// <summary>获取新连接</summary>
        public static SQLiteConnection GetConnection()
        {
            return new SQLiteConnection(_connectionString);
        }
    }
}
```

```csharp
using HwPlcMonitor.Core.Interfaces;
using HwPlcMonitor.Core.Models;
using Dapper;

namespace HwPlcMonitor.Data.Repositories
{
    /// <summary>
    /// PLC数据仓库
    /// 实现IDataService接口，使用Dapper进行数据库操作
    /// </summary>
    public class PlcDataRepository : IDataService
    {
        private readonly ILogger _log = LogCategories.Database;

        /// <summary>保存一条采集数据</summary>
        public async Task<bool> SavePlcDataAsync(PlcDataModel data)
        {
            try
            {
                using (var conn = DatabaseHelper.GetConnection())
                {
                    var sql = @"INSERT INTO plc_data (tag_name, address, value, unit, quality, collect_time)
                               VALUES (@TagName, @Address, @Value, @Unit, @Quality, @CollectTime)";

                    await conn.ExecuteAsync(sql, data);
                }
                return true;
            }
            catch (Exception ex)
            {
                _log.Error(ex, "保存PLC数据失败 {TagName}", data.TagName);
                return false;
            }
        }

        /// <summary>
        /// 批量保存采集数据（使用事务提高性能）
        /// 100条数据在一个事务中插入，比逐条插入快10倍以上
        /// </summary>
        public async Task<bool> SavePlcDataBatchAsync(List<PlcDataModel> dataList)
        {
            if (dataList == null || dataList.Count == 0) return true;

            try
            {
                using (var conn = DatabaseHelper.GetConnection())
                {
                    conn.Open();
                    using (var transaction = conn.BeginTransaction())
                    {
                        try
                        {
                            var sql = @"INSERT INTO plc_data (tag_name, address, value, unit, quality, collect_time)
                                       VALUES (@TagName, @Address, @Value, @Unit, @Quality, @CollectTime)";

                            await conn.ExecuteAsync(sql, dataList, transaction);
                            transaction.Commit();
                            return true;
                        }
                        catch
                        {
                            transaction.Rollback();
                            throw;
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                _log.Error(ex, "批量保存PLC数据失败，数量: {Count}", dataList.Count);
                return false;
            }
        }

        /// <summary>查询指定时间范围的数据</summary>
        public List<PlcDataModel> QueryPlcData(DateTime startTime, DateTime endTime)
        {
            try
            {
                using (var conn = DatabaseHelper.GetConnection())
                {
                    var sql = @"SELECT id, tag_name as TagName, address as Address,
                                       value as Value, unit as Unit, quality as Quality,
                                       collect_time as CollectTime
                                FROM plc_data
                                WHERE collect_time >= @StartTime AND collect_time <= @EndTime
                                ORDER BY collect_time ASC";

                    return conn.Query<PlcDataModel>(sql, new { StartTime, EndTime }).ToList();
                }
            }
            catch (Exception ex)
            {
                _log.Error(ex, "查询PLC数据失败");
                return new List<PlcDataModel>();
            }
        }

        /// <summary>删除指定日期之前的旧数据</summary>
        public async Task<int> DeleteOldRecordsAsync(DateTime beforeDate)
        {
            try
            {
                using (var conn = DatabaseHelper.GetConnection())
                {
                    var sql = "DELETE FROM plc_data WHERE collect_time < @BeforeDate";
                    var count = await conn.ExecuteAsync(sql, new { BeforeDate = beforeDate.ToString("yyyy-MM-dd HH:mm:ss") });
                    _log.Information("清理旧数据 {Count} 条，截止日期: {Date}", count, beforeDate);
                    return count;
                }
            }
            catch (Exception ex)
            {
                _log.Error(ex, "清理旧数据失败");
                return 0;
            }
        }
    }
}
```

### 3.4 报警模块实现

```csharp
// ============================================================
// 文件：AlarmService.cs
// 说明：报警服务 - 检测数据是否超过阈值
// ============================================================
using HwPlcMonitor.Core.Interfaces;
using HwPlcMonitor.Core.Models;
using HwPlcMonitor.Core.Configuration;
using HwPlcMonitor.Core.Logging;

namespace HwPlcMonitor.Core.Services
{
    /// <summary>
    /// 报警服务
    /// 实现IAlarmService接口
    /// 负责报警检测、记录和通知
    /// </summary>
    public class AlarmService : IAlarmService
    {
        private readonly ILogger _log = LogCategories.Alarm;
        private readonly List<AlarmThreshold> _thresholds;
        private readonly IDataService _dataService; // 用于保存报警记录

        // 活跃报警列表（已触发但未恢复）
        private readonly Dictionary<string, AlarmRecord> _activeAlarms = new Dictionary<string, AlarmRecord>();

        // 报警记录列表（用于UI显示）
        private readonly BindingList<AlarmRecord> _recentAlarms = new BindingList<AlarmRecord>();

        public event EventHandler<AlarmRecord> AlarmTriggered;
        public event EventHandler<AlarmRecord> AlarmRecovered;

        public AlarmService(IDataService dataService)
        {
            _dataService = dataService;

            // 从配置加载报警阈值
            var config = ConfigManager.Instance.Settings.Alarm;
            _thresholds = config.Thresholds;
        }

        /// <summary>
        /// 检查数据是否触发报警
        /// </summary>
        public AlarmRecord CheckAlarm(string tagName, double value)
        {
            if (!_thresholds.Any()) return null;

            // 查找该标签的阈值配置
            var threshold = _thresholds.FirstOrDefault(t => t.TagName == tagName);
            if (threshold == null) return null;

            // 检查是否超上限
            if (value > threshold.HighLimit)
            {
                // 如果已有活跃报警，只更新触发值
                if (_activeAlarms.ContainsKey(tagName))
                {
                    _activeAlarms[tagName].TriggerValue = value;
                    return null;
                }

                // 创建新的报警记录
                var alarm = new AlarmRecord
                {
                    TagName = tagName,
                    TriggerValue = value,
                    Threshold = threshold.HighLimit,
                    AlarmTime = DateTime.Now,
                    IsAcknowledged = false,
                    AlarmLevel = threshold.Level,
                    AlarmDescription = $"{tagName}超上限: {value}{threshold.Unit} > {threshold.HighLimit}{threshold.Unit}"
                };

                _activeAlarms[tagName] = alarm;
                _recentAlarms.Insert(0, alarm); // 添加到列表顶部

                // 记录日志
                if (alarm.AlarmLevel == "Critical")
                    _log.Error("触发严重报警: {Alarm}", alarm.AlarmDescription);
                else
                    _log.Warning("触发报警: {Alarm}", alarm.AlarmDescription);

                // 触发事件通知UI
                AlarmTriggered?.Invoke(this, alarm);

                return alarm;
            }
            // 检查是否低于下限
            else if (value < threshold.LowLimit)
            {
                if (_activeAlarms.ContainsKey(tagName))
                {
                    _activeAlarms[tagName].TriggerValue = value;
                    return null;
                }

                var alarm = new AlarmRecord
                {
                    TagName = tagName,
                    TriggerValue = value,
                    Threshold = threshold.LowLimit,
                    AlarmTime = DateTime.Now,
                    IsAcknowledged = false,
                    AlarmLevel = threshold.Level,
                    AlarmDescription = $"{tagName}低于下限: {value}{threshold.Unit} < {threshold.LowLimit}{threshold.Unit}"
                };

                _activeAlarms[tagName] = alarm;
                _recentAlarms.Insert(0, alarm);

                _log.Warning("触发报警: {Alarm}", alarm.AlarmDescription);
                AlarmTriggered?.Invoke(this, alarm);
                return alarm;
            }
            else
            {
                // 数据恢复正常，检查是否有活跃报警需要恢复
                if (_activeAlarms.ContainsKey(tagName))
                {
                    var activeAlarm = _activeAlarms[tagName];
                    activeAlarm.RecoverTime = DateTime.Now;

                    _log.Information("报警恢复: {TagName} 恢复正常", tagName);
                    AlarmRecovered?.Invoke(this, activeAlarm);

                    _activeAlarms.Remove(tagName);
                }

                return null;
            }
        }

        /// <summary>获取所有活跃报警</summary>
        public List<AlarmRecord> GetActiveAlarms()
        {
            return _activeAlarms.Values.ToList();
        }

        /// <summary>确认报警</summary>
        public void AcknowledgeAlarm(int alarmId)
        {
            var alarm = _recentAlarms.FirstOrDefault(a => a.Id == alarmId);
            if (alarm != null)
            {
                alarm.IsAcknowledged = true;
                alarm.AckTime = DateTime.Now;
                _log.Information("报警已确认: {TagName}", alarm.TagName);
            }
        }

        /// <summary>查询历史报警</summary>
        public List<AlarmRecord> QueryAlarmHistory(DateTime startTime, DateTime endTime)
        {
            // 简化实现，实际项目中应从数据库查询
            return _recentAlarms
                .Where(a => a.AlarmTime >= startTime && a.AlarmTime <= endTime)
                .ToList();
        }
    }
```

```csharp
using HwPlcMonitor.Core.Interfaces;
using HwPlcMonitor.Core.Models;

namespace HwPlcMonitor.UI

    public partial class MainForm : Form

        private readonly IDataService _dataService;
        private readonly IAlarmService _alarmService;

        // 采集定时器
        private System.Windows.Forms.Timer _pollTimer;
        private bool _isCollecting = false;

        // 趋势图数据缓存
        private readonly List<KeyValuePair<DateTime, double>> _tempHistory = new List<KeyValuePair<DateTime, double>>();
        private const int MaxChartPoints = 200; // 最多显示200个点

            _alarmService.AlarmRecovered += OnAlarmRecovered;

            // 初始化定时器
            _pollTimer = new System.Windows.Forms.Timer();
            _pollTimer.Tick += async (s, e) => await PollDataAsync();
            _pollTimer.Interval = ConfigManager.Instance.Settings.DataCollect.PollIntervalMs;

            // 初始化UI状态

            if (success)
            {
                UpdateConnectionStatus(true);
                _log.Information("PLC连接成功，可以开始数据采集");
            }
            else
            {
                MessageBox.Show("连接失败，请检查网络和配置", "提示",
                    MessageBoxButtons.OK, MessageBoxIcon.Warning);

            btnConnect.Enabled = true;
        }

        private async void btnDisconnect_Click(object sender, EventArgs e)

            await _commService.DisconnectAsync();
            UpdateConnectionStatus(false);
        }

        // ===== 采集控制 =====

        private void btnStartCollect_Click(object sender, EventArgs e)
        {
            if (!_commService.IsConnected)
            {
                MessageBox.Show("请先连接PLC", "提示", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            _isCollecting = true;
            _pollTimer.Start();
            btnStartCollect.Enabled = false;
            btnStopCollect.Enabled = true;
            UpdateStatusText("数据采集中...");
            _log.Information("开始数据采集，周期: {Interval}ms", _pollTimer.Interval);
        }

        private void btnStopCollect_Click(object sender, EventArgs e)
        {
            StopCollection();
        }

        private void StopCollection()
        {
            _isCollecting = false;
            _pollTimer.Stop();

            // 如果缓冲区还有数据，保存到数据库
            if (_dataBuffer.Count > 0)
            {
                _ = _dataService.SavePlcDataBatchAsync(_dataBuffer);
                _dataBuffer.Clear();
            }

            btnStartCollect.Enabled = true;
            btnStopCollect.Enabled = false;
            UpdateStatusText("采集已停止");
            _log.Information("停止数据采集");
        }

        // ===== 核心采集逻辑 =====

        /// <summary>
        /// 一次采集周期
        /// 读取PLC数据 -> 更新UI -> 缓冲数据 -> 检查报警
        /// </summary>
        private async Task PollDataAsync()
        {
            try
            {
                // 1. 批量读取PLC数据
                var addresses = new List<string> { "D100", "D102", "D104" };
                var values = _commService.ReadMultipleValues(addresses);

                if (values.Count == 0)
                {
                    UpdateStatusText("读取数据为空");
                    return;
                }

                // 2. 更新UI显示（必须检查InvokeRequired）
                if (InvokeRequired)
                {
                    BeginInvoke(new Action(() => UpdateDataDisplay(values)));
                }
                else
                {
                    UpdateDataDisplay(values);
                }

                // 3. 添加到保存缓冲区
                foreach (var kvp in values)
                {
                    var tagConfig = GetTagConfig(kvp.Key);
                    var data = new PlcDataModel
                    {
                        TagName = tagConfig?.TagName ?? kvp.Key,
                        Address = kvp.Key,
                        Value = Convert.ToDouble(kvp.Value),
                        Unit = tagConfig?.Unit ?? "",
                        Quality = 0, // 假设数据质量好
                        CollectTime = DateTime.Now
                    };
                    _dataBuffer.Add(data);
                }

                // 4. 缓冲区达到阈值，批量保存到数据库
                if (_dataBuffer.Count >= BatchSaveThreshold)
                {
                    var batch = _dataBuffer.ToList();
                    _dataBuffer.Clear();
                    _ = _dataService.SavePlcDataBatchAsync(batch);
                }

                // 5. 检查报警
                foreach (var kvp in values)
                {
                    _alarmService.CheckAlarm(kvp.Key, Convert.ToDouble(kvp.Value));
                }

                // 6. 更新趋势图数据
                if (values.TryGetValue("D100", out var temp))
                {
                    if (InvokeRequired)
                        BeginInvoke(new Action(() => UpdateChart(Convert.ToDouble(temp))));
                    else
                        UpdateChart(Convert.ToDouble(temp));
                }
            }
            catch (Exception ex)
            {
                _log.Error(ex, "数据采集异常");
            }
        }

        // ===== UI更新方法 =====

        private void UpdateDataDisplay(Dictionary<string, object> values)
        {
            if (values.TryGetValue("D100", out var temp))
            {
                lblTempValue.Text = $"{temp:0.0} ℃";
                progressBarTemp.Value = Math.Min((int)(Convert.ToDouble(temp) / 100 * 100), 100);
            }
            if (values.TryGetValue("D102", out var pressure))
            {
                lblPressureValue.Text = $"{pressure:0.00} MPa";
            }
            if (values.TryGetValue("D104", out var flow))
            {
                lblFlowValue.Text = $"{flow:0.0} L/min";
            }

            // 更新状态栏的通信信息
            toolStripStatusLabelComm.Text = $"通信正常 | 上次采集: {DateTime.Now:HH:mm:ss.fff}";
        }

        private void UpdateChart(double temperature)
        {
            _tempHistory.Add(new KeyValuePair<DateTime, double>(DateTime.Now, temperature));
            if (_tempHistory.Count > MaxChartPoints)
                _tempHistory.RemoveAt(0);

            // 更新Chart控件
            chartTemp.Series["温度"].Points.DataBindXY(
                _tempHistory.Select(p => p.Key).ToArray(),
                _tempHistory.Select(p => p.Value).ToArray()
            );
        }

        private void UpdateConnectionStatus(bool connected)
        {
            if (InvokeRequired)
            {
                BeginInvoke(new Action(() => UpdateConnectionStatus(connected)));
                return;
            }

            var color = connected ? Color.Green : Color.Red;
            var text = connected ? "已连接" : "未连接";
            lblConnStatus.ForeColor = color;
            lblConnStatus.Text = text;
        }

        // ===== 事件处理 =====

        private void OnConnectionStateChanged(object sender, bool connected)
        {
            if (!connected && _config.AutoReconnect && _isCollecting)
            {
                UpdateStatusText("连接断开，正在尝试重连...");
            }
        }

        private void OnCommunicationError(object sender, string error)
        {
            UpdateStatusText($"通信错误: {error}");
            toolStripStatusLabelComm.Text = $"通信异常: {error}";
        }

        private void OnAlarmTriggered(object sender, AlarmRecord alarm)
        {

        {
            panelAlarm.BackColor = SystemColors.Control;
        }

    }
```

## 四、项目总结

### 4.1 完整数据流

```
PLC设备 --> Modbus TCP通信 --> 数据采集服务 --> UI显示 + 趋势图
                                       |
                                       v
                                 缓冲区（50条）
                                       |
                                       v
                               SQLite数据库存储
                                       |
                                       v
                              报警检测 --> 报警通知 --> UI报警面板
```

### 4.2 关键设计决策

| 决策 | 理由 |
|------|------|
| 使用Modbus TCP | 汇川PLC支持Modbus协议，通用性强 |
| 批量读取寄存器 | 一次请求读取多个地址，减少通信次数 |
| 缓冲区批量保存 | 积累50条数据后批量写入数据库，减少IO次数 |
| 分层架构 | 通信/业务/UI分离，便于维护和测试 |
| 接口抽象 | 便于替换通信实现（如改为Modbus RTU） |
| Serilog结构化日志 | 便于搜索和分析运行日志 |
| JSON配置文件 | 便于外部修改参数，支持热更新 |
