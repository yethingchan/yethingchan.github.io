# 03 - 自动化测试

> 本文讲解上位机项目中的自动化测试技术，包括单元测试、Mock测试和集成测试的实践方法。

---

## 一、知识讲解

### 1.1 为什么上位机需要自动化测试？

| 痛点 | 没有自动化测试 | 有自动化测试 |
|------|---------------|-------------|
| 修改代码后 | 手动测试所有功能，容易遗漏 | 一键运行所有测试，自动发现问题 |
| 通信模块故障 | 需要连接真实PLC才能测试 | 用Mock对象模拟PLC，离线测试 |
| 回归Bug | 修复一个Bug可能引入新Bug | 测试用例确保旧功能正常 |
| 代码重构 | 不敢重构，怕改出问题 | 有测试兜底，放心重构 |

### 1.2 测试类型

| 测试类型 | 测试范围 | 速度 | 维护成本 | 上位机适用场景 |
|----------|----------|------|----------|----------------|
| **单元测试** | 单个方法/类 | 极快（毫秒级） | 低 | 报警判断、数据转换、地址解析 |
| **集成测试** | 多个模块协作 | 快（秒级） | 中 | 通信层 + 数据存储的联调 |
| **手动测试** | 整个系统 | 慢 | 高 | 最终验收、UI交互测试 |

### 1.3 测试框架选型

| 框架 | 说明 | 推荐度 |
|------|------|--------|
| **NUnit** | .NET生态最成熟的测试框架 | 推荐 |
| **xUnit** | 微软官方推荐，开源项目广泛使用 | 推荐 |
| **MSTest** | Visual Studio内置 | 了解 |
| **Moq** | 最流行的Mock框架 | 推荐 |

> **推荐组合**：xUnit + Moq（简洁现代，与Visual Studio集成最好）

```csharp
// ============================================================
// 文件：AddressParserTests.cs
// 说明：地址解析器的单元测试
// 测试Modbus地址解析功能的正确性
// ============================================================
using Xunit;

namespace HwPlcMonitor.Tests.Core
{
    /// <summary>
    /// 地址解析器测试
    /// 测试不同的地址格式能否被正确解析
    /// </summary>
    public class AddressParserTests
    {
        // ===== 基础测试 =====

        /// <summary>
        /// 测试：解析D区地址 "D100" 应返回地址100
        /// [Fact] 标记这是一个测试方法
        /// </summary>
        [Fact]
        public void Parse_D100_ShouldReturnAddress100()
        {
            // Arrange（准备）: 创建测试对象和输入
            var parser = new AddressParser();
            var input = "D100";

            // Act（执行）: 调用被测试的方法
            var result = parser.Parse(input);

            // Assert（断言）: 验证结果是否符合预期
            Assert.Equal(100, result.Address);
            Assert.Equal(0x03, result.FunctionCode); // 功能码应为03（读保持寄存器）
        }

        /// <summary>
        /// 测试：解析 "D0" 应返回地址0
        /// 边界值测试
        /// </summary>
        [Fact]
        public void Parse_D0_ShouldReturnAddress0()
        {
            var parser = new AddressParser();
            var result = parser.Parse("D0");

            Assert.Equal(0, result.Address);
        }

        // ===== 参数化测试 =====

        /// <summary>
        /// 测试：多组D区地址都能被正确解析
        /// [Theory] + [InlineData] 实现参数化测试
        /// 一组测试方法，多组输入数据
        /// </summary>
        [Theory]
        [InlineData("D0", 0)]
        [InlineData("D100", 100)]
        [InlineData("D1000", 1000)]
        [InlineData("D65535", 65535)]
        public void Parse_DAddresses_ShouldReturnCorrectAddress(string input, int expectedAddress)
        {
            var parser = new AddressParser();
            var result = parser.Parse(input);

            Assert.Equal(expectedAddress, result.Address);
        }

        // ===== 异常测试 =====

        /// <summary>
        /// 测试：不支持的地址格式应抛出异常
        /// </summary>
        [Fact]
        public void Parse_UnsupportedAddress_ShouldThrowException()
        {
            var parser = new AddressParser();

            // Assert.Throws 验证是否抛出了预期类型的异常
            var exception = Assert.Throws<ArgumentException>(() => parser.Parse("X100"));

            Assert.Contains("不支持的地址格式", exception.Message);
        }

        [Theory]
        [InlineData("")]
        [InlineData(null)]
        [InlineData("D")]
        [InlineData("DABC")]
        public void Parse_InvalidAddress_ShouldThrowException(string input)
        {
            var parser = new AddressParser();

            Assert.ThrowsAny<Exception>(() => parser.Parse(input));
        }

        // ===== 大小写测试 =====

        [Theory]
        [InlineData("d100", 100)]
        [InlineData("D100", 100)]
        [InlineData("D100", 100)]
        public void Parse_MixedCase_ShouldBeCaseInsensitive(string input, int expected)
        {
            var parser = new AddressParser();
            var result = parser.Parse(input);
            Assert.Equal(expected, result.Address);
        }
    }
}
```

```csharp
// ============================================================

// ============================================================
using Xunit;
using Moq;

namespace HwPlcMonitor.Tests.Core
{
    /// <summary>
    /// 报警服务测试
    /// 使用Moq模拟IDataService，测试报警逻辑的正确性
    /// </summary>
    public class AlarmServiceTests
    {
        /// <summary>
        /// 测试：温度超过上限应触发报警
        /// </summary>
        [Fact]
        public void CheckAlarm_TemperatureExceedsHighLimit_ShouldTriggerAlarm()
        {
            // Arrange: 创建模拟的数据服务
            var mockDataService = new Mock<IDataService>();
            var thresholds = new List<AlarmThreshold>
            {
                new AlarmThreshold
                {
                    TagName = "温度",
                    HighLimit = 85.0,
                    LowLimit = 10.0,
                    Unit = "℃",
                    Level = "Warning"
                }
            };

            // 创建报警服务（传入模拟对象）
            var alarmService = new AlarmService(mockDataService.Object, thresholds);

            // Act: 传入超限的温度值
            var result = alarmService.CheckAlarm("温度", 87.5);

            // Assert: 应触发报警
            Assert.NotNull(result);               // 报警记录不为null
            Assert.Equal("温度", result.TagName);  // 标签名正确
            Assert.Equal(87.5, result.TriggerValue); // 触发值正确
            Assert.Equal(85.0, result.Threshold);  // 阈值正确
            Assert.Equal("Warning", result.AlarmLevel); // 报警级别正确

            // 验证是否触发了报警事件
            var eventTriggered = false;
            alarmService.AlarmTriggered += (s, a) => eventTriggered = true;
            alarmService.CheckAlarm("温度", 90.0);
            Assert.True(eventTriggered);
        }

        /// <summary>
        /// 测试：温度低于下限应触发报警
        /// </summary>
        [Fact]
        public void CheckAlarm_TemperatureBelowLowLimit_ShouldTriggerAlarm()

            {
                new AlarmThreshold
                {
                    TagName = "温度",
                    HighLimit = 85.0,
                    LowLimit = 10.0,
                    Unit = "℃",
                    Level = "Warning"
                }
            };

            var alarmService = new AlarmService(mockDataService.Object, thresholds);

            // 传入低于下限的温度值
            var result = alarmService.CheckAlarm("温度", 5.0);

            Assert.NotNull(result);
            Assert.Equal(10.0, result.Threshold); // 阈值应为下限值
        }

        /// <summary>
        /// 测试：温度在正常范围内不应触发报警
        /// </summary>
        [Fact]
        public void CheckAlarm_TemperatureInRange_ShouldNotTriggerAlarm()
        {
            var mockDataService = new Mock<IDataService>();
            var thresholds = new List<AlarmThreshold>
            {
                new AlarmThreshold
                {
                    TagName = "温度",
                    HighLimit = 85.0,
                    LowLimit = 10.0,
                    Unit = "℃",
                    Level = "Warning"
                }
            };

            var alarmService = new AlarmService(mockDataService.Object, thresholds);

            // 传入正常范围内的温度值
            var result = alarmService.CheckAlarm("温度", 50.0);

            Assert.Null(result); // 不应触发报警
        }

        /// <summary>
        /// 测试：临界值（等于上限）的处理
        /// 根据业务需求，等于上限是否报警需要明确
        /// </summary>
        [Theory]
        [InlineData(85.0, true)]   // 等于上限 -> 应报警
```

```csharp
        }

        /// <summary>
        /// 测试：通信超时时应正确处理异常
        /// </summary>

        public void ReadValue_Timeout_ShouldNotCrash()
        {
            // Arrange: 模拟超时异常
            var mockMaster = new Mock<IModbusMaster>();

                .Setup(m => m.ReadHoldingRegisters(It.IsAny<byte>(), It.IsAny<ushort>(), It.IsAny<ushort>()))

            comm.CommunicationError += (s, e) => errorReceived = true;

            // Act: 尝试读取

            // Assert: 不应崩溃，应触发错误事件，返回空结果

        {

            comm.ConnectionStateChanged += (s, isConnected) => connected = isConnected;
```

## 三、注意事项

### 3.1 单元测试编写原则

| 原则 | 说明 | 示例 |
|------|------|------|
| **AAA模式** | Arrange-Act-Assert三段式 | 先准备、再执行、后验证 |
| **测试独立性** | 每个测试互不影响 | 使用独立的测试数据库 |
| **测试命名** | 方法名描述清楚测试场景 | `CheckAlarm_ExceedsLimit_ShouldTrigger` |
| **一个测试一个断言** | 尽量每个测试只验证一件事 | 不要在一个测试中验证多个不相关的功能 |
| **测试边界值** | 不仅测正常值，还要测边界和异常 | 0、最大值、负数、null |

### 3.2 上位机测试特别注意事项

| 注意点 | 说明 |
|--------|------|
| **不要依赖真实PLC** | 使用Mock模拟通信对象，测试不依赖硬件 |
| **测试数据库用内存或临时文件** | 每个测试使用独立的测试数据库，避免数据污染 |
| **异步代码要用async测试** | xUnit支持async测试方法 |
| **关注报警逻辑** | 报警判断是核心业务逻辑，必须充分测试 |
| **测试错误恢复** | 通信断开、数据库异常等场景的恢复逻辑 |
