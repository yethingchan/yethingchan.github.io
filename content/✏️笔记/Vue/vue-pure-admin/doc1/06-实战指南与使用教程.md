### 一、环境准备

#### 1.1 系统要求

| 工具 | 最低版本 | 推荐版本 | 验证命令 |
|------|----------|----------|----------|

### 二、获取项目并启动

#### 2.1 克隆项目

```bash
git clone https://github.com/pure-admin/vue-pure-admin.git
cd vue-pure-admin
```

#### 2.2 安装依赖

```bash
pnpm install
```

如果安装速度慢，可以切换国内镜像源：

```bash
pnpm install --registry https://registry.npmmirror.com
```

#### 2.3 启动开发服务器

```bash
pnpm dev
```

启动成功后，终端会显示：

```
  VITE v8.x.x  ready in xxx ms

  ➜  Network: http://192.168.x.x:8848/
```

### 三、项目配置详解

#### 3.1 环境变量

vue-pure-admin 使用 `.env` 文件管理环境变量：

```bash
# .env — 所有环境通用
VITE_PUBLIC_PATH=/               # 公共路径
VITE_PORT=8848                    # 开发服务器端口

# .env.development — 开发环境
VITE_PROXY_DOMAIN=/api             # 代理前缀
VITE_PROXY_DOMAIN_REAL=http://localhost:8080  # 后端真实地址
VITE_MOCK=true                     # 是否启用 Mock

# .env.production — 生产环境
VITE_CDN=true                      # 是否启用 CDN 外置
VITE_MOCK=false                    # 生产环境关闭 Mock
```

#### 3.2 代理配置

开发环境中，前端请求 `/api/xxx` 会被代理到后端真实地址：

```typescript
// vite.config.ts 中的代理配置
server: {
  port: 8848,
  proxy: {
    "/api": {
      target: "http://localhost:8080",  // 后端地址
      changeOrigin: true,
      rewrite: path => path.replace(/^\/api/, "")
    }
  }
}
```

如果后端地址变更，修改 `.env.development` 中的 `VITE_PROXY_DOMAIN_REAL` 即可。

#### 3.3 主题配置

启动项目后，点击右上角的**设置图标**，可以调整：

- 主题模式：亮色 / 暗色
- 主题色：自定义主色调
- 布局模式：侧边栏 / 顶部 / 混合
- 侧边栏宽度：自定义
- 标签页：显示 / 隐藏
- 面包屑：显示 / 隐藏

所有配置会自动持久化到 localStorage，刷新后保持。

#### 5.3 创建 API 模块

```typescript
// src/api/device.ts
import { http } from "@/utils/http";
import type { Result, PageResult } from "@/api/types";
import type { DeviceInfo, AlarmRecord, DeviceQuery } from "@/api/types/device";

/** 获取设备列表 */
export const getDeviceListApi = (params?: DeviceQuery) => {
  return http.request<PageResult<DeviceInfo>>(
    "get",
    "/api/device/list",
    { params }
  );
};

/** 获取设备详情 */
export const getDeviceDetailApi = (id: number) => {
  return http.request<DeviceInfo>("get", `/api/device/${id}`);
};

/** 新增设备 */
export const addDeviceApi = (data: Partial<DeviceInfo>) => {
  return http.request<Result<boolean>>("post", "/api/device", { data });
};

/** 修改设备 */
export const updateDeviceApi = (data: DeviceInfo) => {
  return http.request<Result<boolean>>("put", "/api/device", { data });
};

/** 删除设备 */
export const deleteDeviceApi = (id: number) => {
  return http.request<Result<boolean>>("delete", `/api/device/${id}`);
};

/** 获取告警记录 */
export const getAlarmListApi = (params?: {
  deviceId?: number;
  alarmLevel?: string;
  handled?: boolean;
  page?: number;
  pageSize?: number;
}) => {
  return http.request<PageResult<AlarmRecord>>(
    "get",
    "/api/device/alarm",
    { params }
  );
};

/** 处理告警 */
export const handleAlarmApi = (id: number, data: { remark?: string }) => {
  return http.request<Result<boolean>>(
    "put",
    `/api/device/alarm/${id}/handle`,
    { data }
  );
};
```

#### 5.4 创建 Composable

```typescript
// src/composables/useDevice.ts
import { ref } from "vue";
import { useTable } from "@/composables/useTable";
import { getDeviceListApi, deleteDeviceApi } from "@/api/device";
import { ElMessage, ElMessageBox } from "element-plus";
import type { DeviceInfo } from "@/api/types/device";

export function useDevice() {
  const table = useTable<DeviceInfo>(getDeviceListApi);

  /** 删除设备 */
  async function handleDelete(row: DeviceInfo) {
    await ElMessageBox.confirm(
      `确定删除设备「${row.name}」吗？此操作不可恢复。`,
      "提示",
      { type: "warning" }
    );

    const result = await deleteDeviceApi(row.id);
    if (result) {
      ElMessage.success("删除成功");
      table.loadData();
    }
  }

  /** 状态标签颜色 */
  function getStatusType(status: DeviceInfo["status"]) {
    const map = {
      online: "success",
      offline: "info",
      alarm: "danger",
      maintenance: "warning"
    };
    return map[status];
  }

  /** 状态文本 */
  function getStatusText(status: DeviceInfo["status"]) {
    const map = {
      online: "在线",
      offline: "离线",
      alarm: "告警",
      maintenance: "维护中"
    };
    return map[status];
  }

  return {
    ...table,
    handleDelete,
    getStatusType,
    getStatusText
  };
}
```

```typescript
  XLSX.utils.book_append_sheet(workbook, worksheet, "设备列表");
```

#### 6.2 添加 WebSocket 实时监控

```typescript
// composables/useWebSocket.ts
import { ref, onUnmounted } from "vue";

export function useWebSocket(url: string) {
  const data = ref<any>(null);
  const connected = ref(false);
  let ws: WebSocket | null = null;
  let reconnectTimer: number | null = null;

  function connect() {
    ws = new WebSocket(url);

    ws.onopen = () => {
      connected.value = true;
    };

    ws.onmessage = (event) => {
      data.value = JSON.parse(event.data);
    };

    ws.onclose = () => {
      connected.value = false;
      // 5 秒后重连
      reconnectTimer = window.setTimeout(connect, 5000);
    };

    ws.onerror = () => {
      ws?.close();
    };
  }

  function send(message: any) {
    ws?.send(JSON.stringify(message));
  }

  function disconnect() {
    if (reconnectTimer) clearTimeout(reconnectTimer);
    ws?.close();
  }

  connect();
  onUnmounted(disconnect);

  return { data, connected, send, disconnect };
}
```

使用：

```vue
<script setup lang="ts">
import { useWebSocket } from "@/composables/useWebSocket";

const { data: deviceStatus, connected } = useWebSocket(
  "ws://localhost:8080/ws/device-status"
);
</script>

<template>
  <el-tag :type="connected ? 'success' : 'danger'">
    {{ connected ? "已连接" : "未连接" }}
  </el-tag>
  <div v-if="deviceStatus">
    设备 {{ deviceStatus.name }} 当前温度: {{ deviceStatus.temperature }}°C
  </div>
</template>
```

```dockerfile
# 构建阶段
FROM node:22-alpine AS builder
WORKDIR /app
COPY package.json pnpm-lock.yaml ./
RUN corepack enable && pnpm install --frozen-lockfile

RUN pnpm build

FROM nginx:alpine

COPY nginx.conf /etc/nginx/conf.d/default.conf
EXPOSE 80
CMD ["nginx", "-g", "daemon off;"]
```

创建 `nginx.conf`：

```nginx
server {
    listen 80;
    server_name localhost;
    root /usr/share/nginx/html;
    index index.html;

    # SPA 路由支持：所有路径返回 index.html
    location / {
        try_files $uri $uri/ /index.html;
    }

    # API 代理到后端
    location /api/ {
        proxy_pass http://backend:8080/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }

    # 静态资源缓存
    location ~* \.(js|css|png|jpg|svg|woff2)$ {
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # Gzip 压缩
    gzip on;
    gzip_types text/plain text/css application/json application/javascript;
    gzip_min_length 1024;
}
```

构建和运行：

```bash
docker build -t vue-pure-admin .
docker run -d -p 80:80 vue-pure-admin
```

### 八、常见问题与解决方案

#### 8.1 pnpm install 报错

```bash
# 清除缓存重新安装
pnpm store prune
pnpm install

# 如果是 peer dependency 报错
pnpm install --strict-peer-dependencies=false
```

#### 8.2 开发服务器启动后白屏

```bash
# 检查 Node.js 版本
node -v  # 必须 >= 22.22.1

# 清除 Vite 缓存
rm -rf node_modules/.vite
pnpm dev
```

#### 8.3 TypeScript 报错但代码运行正常

通常是类型定义文件未生成。执行：

```bash
# 重新生成自动导入的类型声明
pnpm dev  # 启动后自动生成 auto-imports.d.ts 和 components.d.ts
```

#### 8.4 路由添加后不显示

检查以下几点：

1. 路由文件是否放在 `src/router/modules/` 目录下
2. 路由配置是否使用了 `satisfies RouteConfigsTable`
3. `meta.roles` 是否包含了当前用户的角色
4. `meta.rank` 是否设置（影响菜单排序）
5. 是否需要清除浏览器缓存（动态路由有 `isDynamicLoaded` 标记）

#### 8.5 修改 .env 后不生效

```bash
# 修改 .env 文件后需要重启开发服务器
# 先 Ctrl+C 停止，再重新启动
pnpm dev
```

#### 8.6 代理不生效

```bash
# 检查 .env.development 中的 VITE_PROXY_DOMAIN_REAL 是否正确
# 检查 vite.config.ts 中的 proxy 配置
# 确保后端服务已启动且可访问
curl http://localhost:8080/api/health
```

#### 8.7 Element Plus 组件样式不生效

```typescript
// 确保 main.ts 中正确导入
import "element-plus/dist/index.css";
// 或使用 unplugin-element-plus 自动导入样式
```
