# 11 · 完整综合示例

> 上接：[[00-总览与目录]]
> 本例基于本仓库 `test-module` 的 `LocationsMapper.xml` 改写，覆盖前面各模块知识点。

## 综合 Mapper XML

```xml
<?xml version="1.0" encoding="UTF-8" ?>
<!DOCTYPE mapper
PUBLIC "-//mybatis.org//DTD Mapper 3.0//EN"
"http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="cn.yething.test.mapper.LocationsMapper">

    <!-- resultMap：手动映射（列名 user_name -> userName 等） 见 [[03-resultMap详细]] -->
    <resultMap type="Locations" id="LocationsResult">
        <result property="id"         column="id"/>
        <result property="name"       column="name"/>
        <result property="createTime" column="create_time"/>
    </resultMap>

    <!-- sql 片段复用 见 [[04-sql片段与include]] -->
    <sql id="selectLocationsVo">
        select id, name, create_time from locations
    </sql>

    <!-- 动态条件查询 见 [[05-动态SQL]] -->
    <select id="selectLocationsList" parameterType="Locations" resultMap="LocationsResult">
        <include refid="selectLocationsVo"/>
        <where>
            <if test="name != null and name != ''">
                and name like concat('%', #{name}, '%')
            </if>
            <if test="createTime != null">
                and create_time = #{createTime}
            </if>
        </where>
    </select>

    <select id="selectLocationsById" parameterType="Long" resultMap="LocationsResult">
        <include refid="selectLocationsVo"/>
        where id = #{id}
    </select>

    <!-- 主键自增回填 见 [[07-主键自增与selectKey]] -->
    <insert id="insertLocations" parameterType="Locations"
            useGeneratedKeys="true" keyProperty="id">
        insert into locations
        <trim prefix="(" suffix=")" suffixOverrides=",">
            <if test="name != null and name != ''">name,</if>
            <if test="createTime != null">create_time,</if>
        </trim>
        <trim prefix="values (" suffix=")" suffixOverrides=",">
            <if test="name != null and name != ''">#{name},</if>
            <if test="createTime != null">#{createTime},</if>
        </trim>
    </insert>

    <update id="updateLocations" parameterType="Locations">
        update locations
        <set>
            <if test="name != null and name != ''">name = #{name},</if>
            <if test="createTime != null">create_time = #{createTime},</if>
        </set>
        where id = #{id}
    </update>

    <delete id="deleteLocationsById" parameterType="Long">
        delete from locations where id = #{id}
    </delete>

    <!-- 批量删除：省略 parameterType，数组用 collection="array" 见 [[06-foreach的collection规则]] -->
    <delete id="deleteLocationsByIds">
        delete from locations where id in
        <foreach item="id" collection="array" open="(" separator="," close=")">
            #{id}
        </foreach>
    </delete>

</mapper>
```

## 对应接口（节选）

```java
public interface LocationsMapper {
    List<Locations> selectLocationsList(Locations locations);
    Locations selectLocationsById(Long id);
    int insertLocations(Locations locations);
    int updateLocations(Locations locations);
    int deleteLocationsById(Long id);
    int deleteLocationsByIds(Long[] ids);   // 对应 collection="array"
}
```
