# 02-2 辅助表：字典 / 参数 / 公告

> 上接：[[SpringBoot+Vue3后台搭建/02-数据库与RBAC/01-RBAC核心表SQL]]

## 1) 字典表（前后端下拉不再硬编码）

字典是"键值对配置中心"。比如"性别"类型下有 `0=男 1=女`。前端下拉框、表格标签**统一读字典**，改值不用动代码。

```sql
/* 字典类型（如"系统状态""性别""物料类型"） */
CREATE TABLE `sys_dict_type` (
  `dict_id`    BIGINT NOT NULL AUTO_INCREMENT COMMENT '字典主键',
  `dict_name`  VARCHAR(100) DEFAULT '' COMMENT '字典名称',
  `dict_type`  VARCHAR(100) DEFAULT '' COMMENT '字典类型（唯一键，如 sys_user_sex）',
  `status`      CHAR(1) DEFAULT '0' COMMENT '状态（0正常 1停用）',
  `remark`      VARCHAR(500) DEFAULT '' COMMENT '备注',
  `create_time` DATETIME DEFAULT NULL,
  PRIMARY KEY (`dict_id`),
  UNIQUE KEY `uq_dict_type` (`dict_type`)
) ENGINE=InnoDB COMMENT='字典类型表';

/* 字典数据（类型下的具体键值） */
CREATE TABLE `sys_dict_data` (
  `dict_code`  BIGINT NOT NULL AUTO_INCREMENT COMMENT '字典编码',
  `dict_sort`   INT DEFAULT 0 COMMENT '字典排序',
  `dict_label`  VARCHAR(100) DEFAULT '' COMMENT '字典标签（显示名，如"男"）',
  `dict_value`  VARCHAR(100) DEFAULT '' COMMENT '字典键值（如"0"）',
  `dict_type`   VARCHAR(100) DEFAULT '' COMMENT '字典类型（外键字典类型）',
  `css_class`   VARCHAR(100) DEFAULT NULL COMMENT '样式属性（标签颜色）',
  `list_class`  VARCHAR(100) DEFAULT NULL COMMENT '表格回显样式（primary/success...）',
  `is_default`  CHAR(1) DEFAULT 'N' COMMENT '是否默认（Y/N）',
  `status`      CHAR(1) DEFAULT '0' COMMENT '状态',
  `create_time` DATETIME DEFAULT NULL,
  PRIMARY KEY (`dict_code`)
) ENGINE=InnoDB COMMENT='字典数据表';

-- 示例：性别字典
INSERT INTO sys_dict_type(dict_name,dict_type,status) VALUES ('用户性别','sys_user_sex','0');
INSERT INTO sys_dict_data(dict_sort,dict_label,dict_value,dict_type,list_class)
VALUES (1,'男','0','sys_user_sex','primary'),
       (2,'女','1','sys_user_sex','danger');
```
**讲解**：`sys_dict_data.dict_type` 关联 `sys_dict_type.dict_type`。前端 `getDict('sys_user_sex')` 拿到 `[{label:'男',value:'0'},...]` 直接喂 `<el-select>`。

## 2) 参数配置表（系统全局开关/业务参数）

```sql
CREATE TABLE `sys_config` (
  `config_id`    BIGINT NOT NULL AUTO_INCREMENT COMMENT '参数主键',
  `config_name`   VARCHAR(100) DEFAULT '' COMMENT '参数名称',
  `config_key`    VARCHAR(100) DEFAULT '' COMMENT '参数键名（唯一，如 sys.user.initPassword）',
  `config_value`  VARCHAR(500) DEFAULT '' COMMENT '参数键值',
  `config_type`   CHAR(1)     DEFAULT 'N' COMMENT '系统内置（Y不可删/N可改）',
  `remark`        VARCHAR(500) DEFAULT '' COMMENT '备注',
  `create_time`   DATETIME DEFAULT NULL,
  PRIMARY KEY (`config_id`),
  UNIQUE KEY `uq_config_key` (`config_key`)
) ENGINE=InnoDB COMMENT='参数配置表';

-- 示例业务参数（你的库存项目会用到）
INSERT INTO sys_config(config_name,config_key,config_value,config_type,remark)
VALUES ('用户初始密码','sys.user.initPassword','123456','Y','新增用户默认密码'),
       ('库存预警阈值','wms.stock.warn','10','N','低于该数量红色预警'),
       ('WCS对接地址','wcs.api.url','http://192.168.1.100:8081','N','设备调度系统地址');
```
**讲解**：`config_key` 用点分命名空间（`sys.*`/`wms.*`/`wcs.*`）。后端 `@Value("${wms.stock.warn}")` 或查表读取；前端一般放 `store` 全局。改动后清缓存（[[../07-日志与监控/02-监控与缓存备份]]）。

## 3) 通知公告表（首页弹窗）

```sql
CREATE TABLE `sys_notice` (
  `notice_id`    BIGINT NOT NULL AUTO_INCREMENT COMMENT '公告ID',
  `notice_title`  VARCHAR(100) DEFAULT '' COMMENT '公告标题',
  `notice_type`   CHAR(1) DEFAULT '1' COMMENT '公告类型（1通知 2公告）',
  `notice_content`LONGTEXT COMMENT '公告内容',
  `status`        CHAR(1) DEFAULT '0' COMMENT '状态（0正常 1关闭）',
  `create_by`     VARCHAR(64) DEFAULT '' COMMENT '创建人',
  `create_time`   DATETIME DEFAULT NULL,
  `effect_time`   DATETIME DEFAULT NULL COMMENT '生效时间',
  `top`           CHAR(1) DEFAULT 'N' COMMENT '是否置顶（Y/N）',
  `receive_scope` VARCHAR(100) DEFAULT 'ALL' COMMENT '接收范围（ALL全部/指定role_key）',
  PRIMARY KEY (`notice_id`)
) ENGINE=InnoDB COMMENT='通知公告表';
```
**讲解**：`receive_scope` 控制"哪些角色能看到"。前端登录后拉取"生效中 + 本人角色匹配 + 未读"的公告，弹窗（[[../05-字典与基础数据/02-参数配置与通知公告]] 细讲）。
